import pandas as pd
import numpy as np
import tensorflow as tf

# Load CSV (must have 'Close')
data = pd.read_csv("stock.csv")['Close'].values

# Normalize
data = (data - data.min()) / (data.max() - data.min())

# Create sequences (window = 5)
X, y = [], []
for i in range(len(data) - 5):
    X.append(data[i:i+5])
    y.append(data[i+5])

X = np.array(X).reshape(-1, 5, 1)
y = np.array(y)

# Minimal LSTM model
model = tf.keras.Sequential([
    tf.keras.layers.LSTM(10, input_shape=(5,1)),
    tf.keras.layers.Dense(1)
])

model.compile(optimizer='adam', loss='mse')

# Train
model.fit(X, y, epochs=3)
