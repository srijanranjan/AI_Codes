board = [" "] * 9

def print_board():
    for i in range(0,9,3):
        print(board[i], "|", board[i+1], "|", board[i+2])
        if i < 6: print("--+---+--")
def win(p):
    w = [(0,1,2),(3,4,5),(6,7,8),
         (0,3,6),(1,4,7),(2,5,8),
         (0,4,8),(2,4,6)]
    return any(board[a]==board[b]==board[c]==p for a,b,c in w)

def minimax(maxi):
    if win("O"): return 1
    if win("X"): return -1
    if " " not in board: return 0

    best = -9 if maxi else 9

    for i in range(9):
        if board[i] == " ":
            board[i] = "O" if maxi else "X"
            score = minimax(not maxi)
            board[i] = " "

            best = max(best, score) if maxi else min(best, score)
    return best

def best_move():
    best = -9
    move = -1
    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(False)
            board[i] = " "
            if score > best:
                best = score
                move = i
    board[move] = "O"

while True:
    print_board()
    p = int(input()) - 1

    if board[p] != " ": continue
    board[p] = "X"

    if win("X"): print("Player wins"); break
    if " " not in board: print("Draw"); break

    best_move()

    if win("O"): print("AI wins"); break
    if " " not in board: print("Draw"); break
