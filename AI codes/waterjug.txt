from ollama import chat

def calculator(expression):
    return eval(expression)

def word_count(text):
    return len(text.split())

print("AI Banking Assistant")

while True:
    user = input("You: ")

    if user == "exit":
        break

    if "calculate" in user.lower():
        try:
            expression = input("Enter calculation: ")
            result = calculator(expression)
            print("Agent:", result)
        except:
            print("Invalid calculation")

    elif "count words" in user.lower():
        text = input("Enter text: ")
        print("Agent:", word_count(text))

    else:
        response = chat(
            model="llama3.2:1b",
            messages=[
                {
                    "role": "user",
                    "content": user
                }
            ]
        )

        print("Agent:", response.message.content)
